% keep a heading with at least a few lines (or the start of a table) after it
% \mrfskeep{len}: start a new page unless len remains. Unlike needspace it uses a forced
% break decided from \pagetotal, so it never leaves a preferred break point that the page
% builder could choose after a longtable has started (which repeats the table header).
\makeatletter
\newcommand{\mrfskeep}[1]{\par\penalty\@M\ifdim\dimexpr\pagegoal-\pagetotal\relax<#1\relax\newpage\fi}
\makeatother
